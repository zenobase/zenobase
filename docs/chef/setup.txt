=== Chef Workstation (Ubuntu) ===

# see http://wiki.opscode.com/display/chef/Fast+Start+Guide
sudo aptitude install ruby ruby-dev rubygems libopenssl-ruby rdoc ri irb build-essential wget ssl-cert git-core
sudo gem install chef
git clone git://github.com/opscode/chef-repo.git
mkdir -p ~/.chef
# download knife.rb, ejain.pem, zenobase-validator.pem from https://manage.opscode.com/ to ~/.chef

# create .ssh/config
User ubuntu
IdentityFile /home/ubuntu/.ssh/ejain.pem

# see http://wiki.opscode.com/display/chef/EC2+Bootstrap+Fast+Start+Guide
sudo aptitude install libxslt-dev libxml2-dev

# sudo gem install knife-ec2 --no-rdoc --no-ri
# build patched version of ec2-knife instead (supports setting node attributes)
git clone https://github.com/johntdyer/knife-ec2
cd knife-ec2
gem build knife-ec2.gemspec
sudo gem install knife-ec2-0.5.13.gem
sudo gem cleanup knife-ec2

# install chef-client cookbook
knife cookbook site install chef-client
knife cookbook upload chef-client

# install ntp cookbook
knife cookbook site install ntp
knife cookbook upload ntp

# create elasticsearch cookbook
knife cookbook create elasticsearch
...
knife cookbook upload elasticsearch

# create play cookbook
knife cookbook create play
...
knife cookbook upload play

# create node
knife ec2 server create -I ami-b89842d1 -f m1.small -G zeno --region us-east-1 -Z us-east-1c -x ubuntu -S ejain -r 'recipe[ntp],recipe[elasticsearch],recipe[play]' \
  -j '{ "aws": { "access_key": "AKIAI7MHM3G2FTHQJJ3A", "secret_key": "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" }, "elasticsearch" : { "clusterName" : "zeno-test" }, "play" : { "version" : "1.0.1338397258805" } }'

NODE=

# update
knife ssh name:$NODE 'sudo aptitude -q update && sudo aptitude -yq safe-upgrade && sudo shutdown -r now'

# update node
# knife node run list add $NODE 'recipe[foo]'
# knife ssh name:$NODE 'sudo chef-client'

# check node
knife ssh name:$NODE 'curl localhost:9000/status'
knife ssh name:$NODE 'curl localhost:9200/_cluster/health?pretty=true'

# drop node
yes | knife ec2 server delete $NODE
yes | knife node delete $NODE
